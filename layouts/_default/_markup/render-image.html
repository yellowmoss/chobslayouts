{{- $dest := .Destination -}} {{- $alt := strings.TrimSpace (.Text | default "") -}} {{- $page := .Page -}} {{- /* Helper to store a 320x320 square summary thumbnail once per page */ -}} {{- define "partials/store-summary-thumb.html" -}} {{- $page := .page -}} {{- $res := .res -}} {{- $alt := .alt -}} {{- if and $res (not ($page.Store.Get "summary_thumb")) -}} {{- $sq := $res.Fill "320x320 Smart webp q85" -}} {{- $page.Store.Set "summary_thumb" (dict "src" $sq.RelPermalink "alt" ($alt | default $page.LinkTitle) "width" $sq.Width "height" $sq.Height) -}} {{- end -}} {{- end -}} {{- $isMP3 := strings.HasSuffix (lower $dest) ".mp3" -}} {{- if $isMP3 -}} {{- $src := $dest -}} {{- if not (or (hasPrefix $dest "http://") (hasPrefix $dest "https://") (hasPrefix $dest "/")) -}} {{- with $page.Resources.Get $dest -}} {{- $src = .RelPermalink -}} {{- else -}} {{- with resources.Get $dest -}} {{- $src = .RelPermalink -}} {{- end -}} {{- end -}} {{- end -}} {{- $uid := printf "mp3-%s" (md5 (printf "%s-%s" $page.RelPermalink $src)) -}} {{- return -}} {{- end -}} {{- if or (strings.Contains $dest "vimeo.com/") (strings.Contains $dest "player.vimeo.com/video/") -}} {{- $clean := strings.TrimSuffix "/" (index (split $dest "?") 0) -}} {{- $videoId := index (last 1 (split $clean "/")) 0 -}} {{- $w := "1920" -}} {{- $h := "1080" -}} {{- $title := "sandyride" -}} {{- $match := findRESubmatch `^(\d+)\s*[xX]\s*(\d+)(?:\s+(.*))?$` $alt 1 -}} {{- if gt (len $match) 0 -}} {{- $parts := index $match 0 -}} {{- $w = index $parts 1 -}} {{- $h = index $parts 2 -}} {{- if ge (len $parts) 4 -}} {{- $extra := strings.TrimSpace (index $parts 3) -}} {{- if ne $extra "" -}} {{- $title = $extra -}} {{- end -}} {{- end -}} {{- else if ne $alt "" -}} {{- $title = $alt -}} {{- end -}} {{- /* Fetch Vimeo thumbnail via oEmbed, falling back to Vumbnail */ -}} {{- $watchURL := printf "https://vimeo.com/%s" $videoId -}} {{- $oembedURL := printf "https://vimeo.com/api/oembed.json?url=%s" ($watchURL | urlquery) -}} {{- $thumbURL := printf "https://vumbnail.com/%s.jpg" $videoId -}} {{- with try (resources.GetRemote $oembedURL (dict "key" (printf "vimeo-oembed-%s" $videoId))) -}} {{- if and (not .Err) .Value -}} {{- $meta := .Value.Content | transform.Unmarshal -}} {{- with $meta.thumbnail_url -}} {{- $thumbURL = . -}} {{- end -}} {{- end -}} {{- end -}} {{- with try (resources.GetRemote $thumbURL (dict "key" (printf "vimeo-thumb-%s" $videoId))) -}} {{- if and (not .Err) .Value -}} {{- partial "store-summary-thumb.html" (dict "page" $page "res" .Value "alt" $title) -}} {{- end -}} {{- end -}}
{{- return -}} {{- end -}} {{- $res := $page.Resources.Get $dest -}} {{- if not $res -}} {{- $res = resources.Get (print "/assets/" $dest) -}} {{- end -}} {{- if not $res -}} {{- $res = resources.Get $dest -}} {{- end -}} {{- if $res -}} {{- partial "store-summary-thumb.html" (dict "page" $page "res" $res "alt" $alt) -}} {{- $thumb := "" -}} {{- if gt $res.Width 1000 -}} {{- $thumb = $res.Resize "320x webp q85" -}} {{- else -}} {{- $thumb = $res.Process "webp q85" -}} {{- end -}} {{- $large := "" -}} {{- if gt $res.Width 1600 -}} {{- $large = $res.Resize "1600x webp q90" -}} {{- else -}} {{- $large = $res.Process "webp q90" -}} {{- end -}} {{- else -}} {{- end -}}
